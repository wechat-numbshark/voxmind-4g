cmake_minimum_required(VERSION 3.16)
project(voxmind_ui_check C CXX)
set(LV_CONF_SKIP ON CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_compile_definitions(LV_USE_SNAPSHOT=1 LV_USE_STDLIB_MALLOC=1 LV_USE_OS=0 LV_FONT_FMT_TXT_LARGE=1 LV_USE_FONT_COMPRESSED=1)
get_filename_component(FIRMWARE "${CMAKE_CURRENT_LIST_DIR}/../../../../" ABSOLUTE)
add_subdirectory("${FIRMWARE}/managed_components/lvgl__lvgl" lvgl)
foreach(header esp_err.h esp_lcd_types.h esp_lcd_panel_ops.h esp_app_desc.h esp_check.h esp_log.h esp_lvgl_port.h esp_timer.h)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stubs/${header}" CONTENT "#include \"host.h\"\n")
endforeach()
add_executable(ui-check ui_check.c ../../ui_logic.c ../../fonts/font_cn_16.c ../../fonts/font_cn_24.c ../../fonts/font_num_44.c)
target_include_directories(ui-check PRIVATE . "${CMAKE_CURRENT_BINARY_DIR}/stubs" ../../include "${FIRMWARE}/components/drv_display/include" "${FIRMWARE}/managed_components/lvgl__lvgl/src")
target_compile_options(ui-check PRIVATE -Wno-deprecated-declarations)
target_link_libraries(ui-check PRIVATE lvgl m)
